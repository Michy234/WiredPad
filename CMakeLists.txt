cmake_minimum_required(VERSION 3.20)
project(WiredPad VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HOST REQUIRED IMPORTED_TARGET libusbmuxd-2.0 libimobiledevice-1.0 libavcodec libavutil libswscale)
add_library(platform src/platform.cpp)
target_include_directories(platform PUBLIC src)
add_library(protocol src/protocol.cpp)
target_include_directories(protocol PUBLIC src)
target_compile_options(protocol PRIVATE -Wall -Wextra -Wpedantic)
add_executable(wiredpad-host src/main.cpp src/transport.cpp src/display.cpp src/encoder.cpp src/input.cpp)
target_include_directories(wiredpad-host PRIVATE third_party)
target_link_libraries(wiredpad-host PRIVATE protocol platform PkgConfig::HOST ${CMAKE_DL_LIBS})
target_compile_options(wiredpad-host PRIVATE -Wall -Wextra -Wpedantic)
option(WIREDPAD_BUILD_APP "Build the Qt desktop app" ON)
if(WIREDPAD_BUILD_APP)
    find_package(Qt6 REQUIRED COMPONENTS Widgets DBus)
    add_executable(wiredpad app/main.cpp app/window.cpp)
    qt_add_resources(wiredpad wiredpad_assets PREFIX "/" FILES assets/icon.svg)
    target_link_libraries(wiredpad PRIVATE Qt6::Widgets Qt6::DBus platform)
    target_compile_options(wiredpad PRIVATE -Wall -Wextra -Wpedantic)
    install(TARGETS wiredpad RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(BUILD_TESTING)
    add_executable(protocol_test tests/protocol_test.cpp src/transport.cpp src/input.cpp)
    target_link_libraries(protocol_test PRIVATE protocol PkgConfig::HOST)
    add_test(NAME protocol COMMAND protocol_test)
    add_executable(platform_test tests/platform_test.cpp)
    target_link_libraries(platform_test PRIVATE platform)
    add_test(NAME platform COMMAND platform_test)
    add_executable(encoder_test tests/encoder_test.cpp src/encoder.cpp)
    target_link_libraries(encoder_test PRIVATE protocol PkgConfig::HOST)
    add_test(NAME encoder COMMAND encoder_test)
endif()
install(TARGETS wiredpad-host RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS scripts/map-input-plasma.sh DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME wiredpad-map-input)
install(FILES packaging/io.github.Michy234.WiredPad.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/icon.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME io.github.Michy234.WiredPad.svg)
install(FILES packaging/io.github.Michy234.WiredPad.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
install(FILES packaging/wiredpad.service DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/user)
install(FILES packaging/70-wiredpad-uinput.rules DESTINATION ${CMAKE_INSTALL_LIBDIR}/udev/rules.d)
install(FILES packaging/wiredpad-modules.conf DESTINATION ${CMAKE_INSTALL_LIBDIR}/modules-load.d RENAME wiredpad.conf)
install(FILES README.md PROTOCOL_IOS.md docs/setup.md docs/testing.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES LICENSE third_party/EVDI-LICENSE third_party/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/wiredpad)
